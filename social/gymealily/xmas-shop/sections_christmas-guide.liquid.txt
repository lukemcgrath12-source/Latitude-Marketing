{%- liquid
  assign s = section.settings
  assign src = collections[s.collection_handle]
  if src == blank or src.products_count == 0
    assign src = collections['christmas']
  endif
-%}
{%- paginate src.products by 250 -%}
{%- assign pool = src.products | sort: 'created_at' | reverse -%}
<div class="glx">
{%- comment -%} GIFT GUIDE {%- endcomment -%}
  {%- if s.guide_show -%}
  <section class="glx-sec" id="glx-guide" aria-labelledby="glx-guide-h">
    <div class="glx-w">
      <div class="glx-head glx-head--center">
        {%- if s.guide_kicker != blank -%}<p class="glx-k">{{ s.guide_kicker }}</p>{%- endif -%}
        <h2 class="glx-h" id="glx-guide-h">{{ s.guide_heading }}</h2>
        {%- if s.guide_text != blank -%}<p class="glx-p">{{ s.guide_text }}</p>{%- endif -%}
      </div>
      <div class="glx-guide">
        {%- for i in (1..4) -%}
          {%- liquid
            assign t_key = 'tile' | append: i | append: '_title'
            assign title = s[t_key]
            if title == blank
              continue
            endif
            assign d_key = 'tile' | append: i | append: '_text'
            assign u_key = 'tile' | append: i | append: '_url'
            assign k_key = 'tile' | append: i | append: '_keywords'
            assign m_key = 'tile' | append: i | append: '_max'
            assign im_key = 'tile' | append: i | append: '_image'
            assign kws = s[k_key] | downcase | split: ','
            assign maxp = s[m_key] | times: 100
            assign tile_img = blank
            unless s[im_key] != blank
              for p in pool
                unless p.available and p.featured_media
                  continue
                endunless
                if maxp > 0 and p.price > maxp
                  continue
                endif
                assign tagstr = p.tags | join: ' '
                assign hay = p.title | append: ' ' | append: p.type | append: ' ' | append: p.vendor | append: ' ' | append: tagstr | downcase
                assign hit = false
                if kws.size == 0
                  assign hit = true
                endif
                for kw in kws
                  assign kwt = kw | strip
                  if kwt != blank and hay contains kwt
                    assign hit = true
                    break
                  endif
                endfor
                if hit
                  assign tile_img = p.featured_media
                  break
                endif
              endfor
            endunless
          -%}
          <a class="glx-tile" href="{{ s[u_key] | default: src.url }}">
            <div class="glx-tile__img">
              {%- if s[im_key] != blank -%}
                {{ s[im_key] | image_url: width: 600 | image_tag: loading: 'lazy', class: 'cover', alt: title }}
              {%- elsif tile_img -%}
                {{ tile_img | image_url: width: 600 | image_tag: loading: 'lazy', alt: title }}
              {%- endif -%}
            </div>
            <div class="glx-tile__txt">
              <div class="glx-tile__t">{{ title }}</div>
              {%- if s[d_key] != blank -%}<div class="glx-tile__d">{{ s[d_key] }}</div>{%- endif -%}
              <span class="glx-tile__go">Shop now</span>
            </div>
          </a>
        {%- endfor -%}
      </div>
    </div>
  </section>
  {%- endif -%}

  </div>
{%- endpaginate -%}
