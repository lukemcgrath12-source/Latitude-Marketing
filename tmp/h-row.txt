  assign src = collections[s.collection_handle]

<div class="index-section{% if s.alt %} index-section--alt{% endif %}">

          {%- assign n = 0 -%}
          {%- assign vendors_used = '|' -%}
          {%- for p in src.products -%}

              {%- assign skip = false -%}

              {%- if skip -%}{%- continue -%}{%- endif -%}
            {%- endif -%}
            {%- if s.one_per_brand -%}
              {%- assign vkey = '|' | append: p.vendor | append: '|' -%}
              {%- if vendors_used contains vkey -%}{%- continue -%}{%- endif -%}
              {%- assign vendors_used = vendors_used | append: p.vendor | append: '|' -%}
            {%- endif -%}
            {%- assign n = n | plus: 1 -%}
            {%- render 'product-grid-item', product: p, grid_item_width: 'small--one-half medium-up--one-quarter' -%}
          {%- endfor -%}
        {%- endpaginate -%}
